// ===== Генератор извилистых дендритов ===== (function buildDendrites() { const g = document.getElementById('dendrites'); if (!g) return; const SVG_NS = 'http://www.w3.org/2000/svg'; const somaX = 120, somaY = 80; // центр тела const somaR = 34; // радиус тела по X // Псевдослучайный генератор с фиксированным seed — чтобы рисунок // был одинаковым при каждой загрузке страницы let seed = 42; function rand() { seed = (seed * 1664525 + 1013904223) % 4294967296; return seed / 4294967296; } function randRange(a, b) { return a + (b - a) * rand(); } // Рекурсивно рисует ветку от точки (x, y) в направлении angle function branch(x, y, angle, length, width, depth) { if (depth <= 0 || length < 4) return; // Конечная точка с небольшим искривлением const bend = randRange(-0.5, 0.5); // изгиб const endAngle = angle + bend; const ex = x + Math.cos(endAngle) * length; const ey = y + Math.sin(endAngle) * length; // Контрольные точки для кубической Безье — дают извилистость const c1x = x + Math.cos(angle + randRange(-0.4, 0.4)) * length * 0.4; const c1y = y + Math.sin(angle + randRange(-0.4, 0.4)) * length * 0.4; const c2x = x + Math.cos(endAngle + randRange(-0.4, 0.4)) * length * 0.6; const c2y = y + Math.sin(endAngle + randRange(-0.4, 0.4)) * length * 0.6; const path = document.createElementNS(SVG_NS, 'path'); path.setAttribute('class', 'neuron-dendrite'); path.setAttribute('d', `M ${x.toFixed(1)} ${y.toFixed(1)} ` + `C ${c1x.toFixed(1)} ${c1y.toFixed(1)}, ` + `${c2x.toFixed(1)} ${c2y.toFixed(1)}, ` + `${ex.toFixed(1)} ${ey.toFixed(1)}` ); path.setAttribute('stroke-width', width.toFixed(2)); g.appendChild(path); // Ветвление: 2 потомка с разными углами const nChildren = depth > 2 ? 2 : (rand() < 0.6 ? 2 : 1); for (let i = 0; i < nChildren; i++) { const spread = randRange(0.35, 0.75) * (i === 0 ? -1 : 1); branch( ex, ey, endAngle + spread, length * randRange(0.6, 0.8), width * 0.7, depth - 1 ); } } // Три «ствола» дендритов, выходящих из тела в разные стороны const trunks = [ { angle: Math.PI * 0.95, length: 26, width: 2.4, depth: 4 }, // влево-вверх { angle: Math.PI * 1.05, length: 30, width: 2.6, depth: 5 }, // влево { angle: Math.PI * 1.15, length: 24, width: 2.2, depth: 4 } // влево-вниз ]; for (const t of trunks) { // Стартовая точка — на границе эллипса тела в направлении angle const sx = somaX + Math.cos(t.angle) * somaR; const sy = somaY + Math.sin(t.angle) * (somaR * 0.88); // тело чуть сплюснуто branch(sx, sy, t.angle, t.length, t.width, t.depth); } })();
